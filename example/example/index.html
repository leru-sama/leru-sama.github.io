代码块：

//居然可以支持verilog！
module hello_world (
    input clk,
    input rst_n,
    output reg [7:0] led
);

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        led <= 8'b00000000;
    end else begin
        led <= led + 1'b1;
    end
end

endmodule
